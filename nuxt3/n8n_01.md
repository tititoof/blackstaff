```javascript
const fs = require('fs');
const path = require('path');

const { project, instruction, files, mode, basePath } = $input.first().json;

function readFileSafe(filePath, max = 3000) {
  try { return fs.readFileSync(filePath, 'utf8').slice(0, max); }
  catch { return null; }
}

function scanStructure(dir, depth = 0) {
  if (depth > 3) return '';
  const ignored = ['node_modules', 'vendor', '.git', 'dist', 'build', '.nuxt', '.output', '__pycache__'];
  try {
    return fs.readdirSync(dir, { withFileTypes: true })
      .filter(e => !ignored.includes(e.name) && !e.name.startsWith('.'))
      .map(e => {
        const indent = '  '.repeat(depth);
        return e.isDirectory()
          ? `${indent}${e.name}/\n` + scanStructure(path.join(dir, e.name), depth + 1)
          : `${indent}${e.name}\n`;
      }).join('');
  } catch { return ''; }
}

const structure = scanStructure(basePath);
const customCtx = readFileSafe(path.join(basePath, '.blackstaff-context.md')) || '';
const pkg = JSON.parse(readFileSafe(path.join(basePath, 'package.json')) || '{}');
const deps = Object.keys({ ...pkg.dependencies, ...pkg.devDependencies }).join(', ');

let filesContent = '';
for (const f of files) {
  const content = readFileSafe(path.join(basePath, f));
  if (content) {
    filesContent += `\n### Fichier existant : ${f}\n\`\`\`typescript\n${content}\n\`\`\`\n`;
  }
}

return [{
  json: {
    project,
    instruction,
    files,
    mode,
    basePath,
    structure,
    customCtx,
    filesContent,
    deps
  }
}];
```

## NŒUD 6 — Parse Generated Code
```javascript
const generatedText = $input.first().json.text;
const basePath = $input.first().json.basePath;
const instruction = $input.first().json.instruction;
const project = $input.first().json.project;

const fileRegex = /```filepath:([^\n]+)\n([\s\S]*?)```/g;
const filesToWrite = [];
let match;

while ((match = fileRegex.exec(generatedText)) !== null) {
  const relativePath = match[1].trim();
  const content = match[2];
  filesToWrite.push({
    relativePath,
    absolutePath: `${basePath}/${relativePath}`,
    content
  });
}

if (filesToWrite.length === 0) {
  return [{
    json: {
      parseError: true,
      message: 'Aucun bloc filepath détecté — BLACKSTAFF n\'a pas suivi le format attendu',
      rawResponse: generatedText,
      basePath,
      instruction,
      project
    }
  }];
}

return [{
  json: {
    filesToWrite,
    fileCount: filesToWrite.length,
    fileList: filesToWrite.map(f => f.relativePath).join(', '),
    generatedText,
    basePath,
    instruction,
    project,
    parseError: false
  }
}];
```

## NŒUD 9 — Write Files
```javascript
const fs = require('fs');
const path = require('path');

const verification = $input.first().json.text;
const { filesToWrite, basePath, generatedText, instruction, project, fileList } = $input.first().json;

// Bloquer l'écriture si REFUSER
if (verification.includes('REFUSER')) {
  return [{
    json: {
      writeBlocked: true,
      reason: 'BLACKSTAFF a refusé d\'écrire — la vérification a échoué',
      verification,
      generatedText,
      instruction,
      project,
      basePath,
      written: [],
      errors: []
    }
  }];
}

const written = [];
const errors = [];

for (const file of filesToWrite) {
  try {
    const dir = path.dirname(file.absolutePath);
    fs.mkdirSync(dir, { recursive: true });
    fs.writeFileSync(file.absolutePath, file.content, 'utf8');
    written.push(file.relativePath);
  } catch (err) {
    errors.push({ file: file.relativePath, error: err.message });
  }
}

return [{
  json: {
    writeBlocked: false,
    written,
    errors,
    writtenCount: written.length,
    errorCount: errors.length,
    verification,
    generatedText,
    instruction,
    project,
    basePath
  }
}];
```

## NŒUD 10 — Run Lint & Typecheck
```javascript
const { execSync } = require('child_process');

const { written, errors, writeBlocked, verification, generatedText, instruction, project, basePath } = $input.first().json;

function run(cmd, cwd) {
  try {
    const output = execSync(cmd, {
      cwd,
      timeout: 60000,
      encoding: 'utf8',
      stdio: 'pipe'
    });
    return { success: true, output: output.slice(0, 2000) };
  } catch (err) {
    return {
      success: false,
      output: ((err.stdout || '') + (err.stderr || '')).slice(0, 2000)
    };
  }
}

let lint = { success: true, output: 'Écriture bloquée — lint ignoré' };
let typecheck = { success: true, output: 'Écriture bloquée — typecheck ignoré' };

if (!writeBlocked && written.length > 0) {
  lint = run('pnpm lint --reporter=compact 2>&1 || true', basePath);
  typecheck = run('pnpm typecheck 2>&1 || true', basePath);
}

return [{
  json: {
    lint,
    typecheck,
    written,
    errors,
    writeBlocked,
    verification,
    generatedText,
    instruction,
    project,
    basePath
  }
}];
```

```javascript
${verification}

---

## Code généré
${generatedText}

---
*BLACKSTAFF — qwen3.5:latest via Ollama — Waterdeep Dev Guild*`;

  json: {
    output: response,
    success: !writeBlocked && written.length > 0 && lint.success && typecheck.success,
```
