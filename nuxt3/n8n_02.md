## PRINCIPE DE LA MÉMOIRE

# Chaque échange est sauvegardé dans .blackstaff/memory.jsonl
# Format : une ligne JSON par échange
# {"ts":"2026-05-01T10:00:00Z","mode":"generate","instruction":"...","filesWritten":["..."],"response":"...","status":"VALIDÉ"}
#
# À chaque appel :
#   1. Read Context lit les 20 derniers échanges
#   2. Build Prompt les injecte dans le contexte Ollama
#   3. Format Response écrit le nouvel échange dans le fichier

## NŒUD 3 — Read Blackstaff Config (mis à jour avec mémoire)
```javascript
const fs   = require('fs');
const path = require('path');

const {
  project, instruction, files, mode,
  basePath, blackstaffPath
} = $input.first().json;

// ── Helpers ──────────────────────────────────────────────────────

function readFileSafe(filePath, max = 8000) {
  try   { return fs.readFileSync(filePath, 'utf8').slice(0, max); }
  catch { return null; }
}

function scanStructure(dir, depth = 0) {
  if (depth > 3) return '';
  const ignored = [
    'node_modules', 'vendor', '.git', 'dist', 'build',
    '.nuxt', '.output', '__pycache__', 'tmp', 'log',
    'storage', 'coverage', 'target', '.next', '.svelte-kit'
  ];
  try {
    return fs.readdirSync(dir, { withFileTypes: true })
      .filter(e => !ignored.includes(e.name) && !e.name.startsWith('.'))
      .map(e => {
        const indent = '  '.repeat(depth);
        return e.isDirectory()
          ? `${indent}${e.name}/\n` + scanStructure(path.join(dir, e.name), depth + 1)
          : `${indent}${e.name}\n`;
      }).join('');
  } catch { return ''; }
}

// ── Lire la config .blackstaff/ ───────────────────────────────────

const contextMd      = readFileSafe(path.join(blackstaffPath, 'context.md'));
const systemPromptMd = readFileSafe(path.join(blackstaffPath, 'system-prompt.md'));
const commandsMd     = readFileSafe(path.join(blackstaffPath, 'commands.md'));

if (!contextMd || !systemPromptMd || !commandsMd) {
  const missing = [
    !contextMd      && 'context.md',
    !systemPromptMd && 'system-prompt.md',
    !commandsMd     && 'commands.md'
  ].filter(Boolean).join(', ');
  throw new Error(`[BLACKSTAFF] Fichiers manquants dans .blackstaff/ : ${missing}`);
}

// ── Parser commands.md ────────────────────────────────────────────

const commands = {};
commandsMd.split('\n').forEach(line => {
  const match = line.match(/^(\w+)\s*:\s*(.+)$/);
  if (match) commands[match[1].trim()] = match[2].trim();
});

// ── Lire la mémoire (20 derniers échanges) ────────────────────────

const memoryPath = path.join(blackstaffPath, 'memory.jsonl');
let memory = [];

try {
  const raw = fs.readFileSync(memoryPath, 'utf8');
  memory = raw
    .split('\n')
    .filter(line => line.trim().length > 0)
    .map(line => {
      try { return JSON.parse(line); }
      catch { return null; }
    })
    .filter(Boolean)
    .slice(-20); // 20 derniers échanges
} catch {
  // Pas encore de mémoire — premier échange
  memory = [];
}

// ── Structure du projet ───────────────────────────────────────────

const structure = scanStructure(basePath);

// ── Fichiers demandés ─────────────────────────────────────────────

let filesContent = '';
for (const f of files) {
  const content = readFileSafe(path.join(basePath, f));
  if (content) {
    const ext = path.extname(f).slice(1) || 'text';
    filesContent += `\n### ${f}\n\`\`\`${ext}\n${content}\n\`\`\`\n`;
  }
}

return [{
  json: {
    project,
    instruction,
    files,
    mode,
    basePath,
    blackstaffPath,
    contextMd,
    systemPromptMd,
    commandsMd,
    commands,
    structure,
    filesContent,
    memory,
    memoryPath
  }
}];
```

## NŒUD 4 — Build Prompt (mis à jour avec mémoire)
```javascript
const {
  project, instruction, mode,
  contextMd, systemPromptMd,
  structure, filesContent,
  memory, memoryPath,
  basePath, commands
} = $input.first().json;

// ── Formater la mémoire pour le contexte ─────────────────────────

let memorySection = '';

if (memory.length > 0) {
  const formatted = memory.map((entry, i) => {
    const date    = new Date(entry.ts).toLocaleString('fr-FR');
    const files   = entry.filesWritten && entry.filesWritten.length > 0
      ? `\nFichiers : ${entry.filesWritten.join(', ')}`
      : '';
    const status  = entry.status ? ` [${entry.status}]` : '';

    return `### Échange ${i + 1} — ${date}${status}
Mode : ${entry.mode}
Instruction : ${entry.instruction}${files}
Résumé : ${entry.summary || entry.response.slice(0, 300)}`;
  }).join('\n\n');

  memorySection = `## Mémoire des échanges précédents (${memory.length} derniers)
Ces échanges sont le contexte de travail accumulé sur ce projet.
Utilise-les pour maintenir la cohérence, éviter les contradictions,
et t'appuyer sur les décisions déjà prises.

${formatted}

---`;
}

// ── Instructions par mode ─────────────────────────────────────────

const modeInstructions = {
  generate: `## Mode : GÉNÉRATION
Tu dois CRÉER du code nouveau.
Pour chaque fichier à créer, utilise OBLIGATOIREMENT ce format :

\`\`\`filepath:chemin/relatif/depuis/racine
contenu du fichier
\`\`\`

- Chaque fichier a son propre bloc filepath
- Le chemin est relatif à la racine du projet
- Sans filepath, le fichier ne sera pas écrit sur le disque
- Génère TOUS les fichiers nécessaires pour que la feature soit complète`,

  modify: `## Mode : MODIFICATION
Tu dois MODIFIER du code existant.
Pour chaque fichier modifié, utilise le format filepath avec le contenu COMPLET.
Montre un diff AVANT / APRÈS pour chaque modification.`,

  explain: `## Mode : EXPLICATION
Tu dois EXPLIQUER le code sans le modifier.
Pas de bloc filepath nécessaire.
Analyse en profondeur, identifie les problèmes, propose des améliorations.`
};

// ── Assembly ──────────────────────────────────────────────────────

const systemPrompt = `${systemPromptMd}

## Contexte du projet
${contextMd}

${memorySection}

## Règle de format — CRITIQUE
${modeInstructions[mode] || modeInstructions.generate}`;

const userPrompt = `## Instruction
${instruction}

## Structure actuelle du projet
\`\`\`
${structure}
\`\`\`

${filesContent ? `## Fichiers concernés\n${filesContent}` : ''}

Réalise l'instruction complètement.
Pour chaque fichier généré, explique brièvement le choix d'implémentation.
Si cette demande est liée à un échange précédent, mentionne-le explicitement.`;

return [{
  json: {
    systemPrompt,
    userPrompt,
    instruction,
    mode,
    project,
    basePath,
    commands,
    contextMd,
    memoryPath,
    memory
  }
}];
```

## NŒUD 11 — Format Response (mis à jour : sauvegarde mémoire)
```javascript
const fs = require('fs');

const {
  written, errors, writeBlocked,
  results, verification, generatedText,
  instruction, project, mode,
  memoryPath, memory
} = $input.first().json;

// ── Statut vérification ───────────────────────────────────────────

let statusBadge = '📋 ANALYSÉ';
if (verification.includes('VALIDÉ'))           statusBadge = '✅ VALIDÉ';
else if (verification.includes('À AMÉLIORER')) statusBadge = '⚠️ À AMÉLIORER';
else if (verification.includes('REFUSER'))     statusBadge = '❌ REFUSÉ';

const scoreMatch = verification.match(/Score global\s*:\s*(\d+)\/40/i);
const score = scoreMatch ? `${scoreMatch[1]}/40` : 'N/A';

// ── Fichiers ──────────────────────────────────────────────────────

const filesSection = writeBlocked
  ? `⚔️ Écriture bloquée — vérification échouée`
  : mode === 'explain'
    ? `ℹ️ Mode explain — aucun fichier écrit`
    : written.length > 0
      ? written.map(f => `  ✅ ${f}`).join('\n')
      : '  ⚠️ Aucun fichier écrit';

const errorsSection = errors.length > 0
  ? '\n\n### Erreurs\n' + errors.map(e => `  ❌ ${e.file} — ${e.error}`).join('\n')
  : '';

// ── Commandes ─────────────────────────────────────────────────────

let commandsSection = '';
for (const [name, result] of Object.entries(results || {})) {
  const icon = result.success ? '✅' : '⚠️';
  commandsSection += `\n${icon} **${name}** : \`${result.cmd}\``;
  if (!result.success) {
    commandsSection += `\n\`\`\`\n${result.output}\n\`\`\``;
  }
}

// ── Générer un résumé court pour la mémoire ───────────────────────

const summary = [
  `Mode : ${mode}`,
  written.length > 0 ? `Fichiers : ${written.join(', ')}` : null,
  `Statut : ${statusBadge}`,
  `Score : ${score}`
].filter(Boolean).join(' | ');

// ── Sauvegarder dans memory.jsonl ─────────────────────────────────

const memoryEntry = {
  ts:           new Date().toISOString(),
  mode,
  instruction,
  filesWritten: written || [],
  status:       statusBadge,
  score,
  summary,
  response:     generatedText.slice(0, 500) // extrait pour ne pas gonfler le fichier
};

try {
  fs.appendFileSync(memoryPath, JSON.stringify(memoryEntry) + '\n', 'utf8');
} catch (err) {
  // Ne pas bloquer la réponse si la sauvegarde échoue
  console.error('[BLACKSTAFF] Erreur sauvegarde mémoire :', err.message);
}

// ── Réponse finale ────────────────────────────────────────────────

const allSuccess = !writeBlocked
  && (mode === 'explain' || written.length > 0)
  && Object.values(results || {}).every(r => r.success);

const memoryInfo = `_Échange #${(memory || []).length + 1} sauvegardé dans la mémoire du projet_`;

const response = `# ⚔️ BLACKSTAFF — ${project}
*Khelben Arunsun, Archimage de Waterdeep*

## Instruction
> ${instruction}

## Statut : ${statusBadge} — Score : ${score}

## Fichiers
${filesSection}${errorsSection}

## Vérifications
${commandsSection || '_Aucune commande exécutée_'}

---

## Rapport de l'archimage
${verification}

---

## Code généré
${generatedText}

---
${memoryInfo}
*BLACKSTAFF — qwen3.5:latest — Waterdeep Dev Guild*`;

return [{
  json: {
    output: response,
    success: allSuccess,
    written,
    errors,
    statusBadge,
    score
  }
}];
```
